# 塔筒系统 · 月份切换 + 计划完成导出 + 三页联动（最终版提示词）

> 系统：Vue3 + Pinia + Element Plus + Vue Router（前端）/ FastAPI + MySQL（后端）。
> 范围：生产进度总览、排产计划总览、出品排名总览三个子页面，共用同一"月份"筛选维度；并支持导出当前月计划完成情况。
> 模式：仅前端预设账号权限（本提示词不涉及登录改造，登录按既有提示词执行）。

---

## ⚠️ 关键前提：月份归属口径

`projects` 表**没有独立的月份字段**。统一采用 **`created_at` 的年月**作为「调度令月份」：
- 格式 `'YYYY-MM'`（如 `2026-08`）。
- 月度调度令导入 / 手动创建的项目，`created_at` 即归入该月。
- 月份筛选 = 按 `created_at` 前 7 位等于所选月份过滤。
- 默认月份 = 当前系统时间的年月。
- **三页（生产进度总览 / 排产计划总览 / 出品排名总览）共用同一个月份值**，任一页切换即全局同步。

### 1. `backend/app/core/db.py` — `get_projects_filtered` 增加 month 参数

```python
def get_projects_filtered(keyword: str | None = None, person: str | None = None,
                          status: str | None = None, skip: int = 0, limit: int = 10,
                          month: str | None = None):          # ← 新增 month
    rows = get_all_projects(None)

    # ★ 按调度令月份（created_at 年月）过滤
    if month:
        rows = [p for p in rows if str(p.get("created_at", ""))[:7] == month]

    if keyword:
        kw = str(keyword).strip().lower()
        rows = [
            p for p in rows
            if kw in str(p.get("project_name", "")).lower()
            or kw in str(p.get("machine_type", "")).lower()
        ]
    if person:
        pname = str(person).strip()
        rows = [p for p in rows if str(p.get("delivery_person", "")) == pname]

    # —— 以下 enrich / risk_level / progress_pct 计算逻辑保持不变 ——
    from ..services.node_service import enrich_rows
    today_s = str(date.today())
    all_pids = [p["id"] for p in rows]
    plans_map = get_node_plans_batch(all_pids)
    actuals_map = get_node_actuals_batch(all_pids)
    conn = get_connection()
    try:
        with conn.cursor() as cur:
            cur.execute("SELECT DISTINCT project_id FROM process_node_plans")
            has_schedule_ids = {r["project_id"] for r in cur.fetchall()}
    finally:
        conn.close()
    for p in rows:
        pid = p["id"]
        p["has_schedule_plan"] = pid in has_schedule_ids
        plans = plans_map.get(pid, [])
        actuals = actuals_map.get(pid, {})
        nodes = enrich_rows(plans, actuals)
        has_overdue = any(r["status"] == "overdue" for r in nodes)
        has_today_unfinished = any(
            r["status"] == "in_progress"
            and str(r["plan_date"])[:10] == today_s
            and r["actual_qty"] < r["plan_qty"]
            for r in nodes
        )
        p["risk_level"] = "delayed" if has_overdue else ("warning" if has_today_unfinished else "normal")
        att_plans = [n for n in plans if n["process_name"] == "附件安装"]
        att_plan_qty = sum(int(n["plan_qty"] or 0) for n in att_plans)
        att_actual_qty = sum(int(actuals.get(n["id"], {}).get("actual_qty", 0) or 0) for n in att_plans)
        p["progress_pct"] = float(round(att_actual_qty / att_plan_qty * 100, 1)) if att_plan_qty else 0.0

    if status in ("normal", "warning", "delayed"):
        rows = [p for p in rows if p.get("risk_level") == status]

    total = len(rows)
    items = rows[skip: skip + limit]
    return items, total
```

> 确保 `get_all_projects` 返回的字典包含 `created_at` 字段。

### 5. 出品排名 — 统一为 `created_at` 月份口径（推荐一致）

> 排名页原按 `process_node_plans.plan_date` 月份归类项目；为"与其他页面一致"，改为按 `created_at` 月份筛选项目集合，累计计划/完成仍按该月内「附件安装」节点汇总。

在 `ranking_service.py` 的 `get_production_ranking` 中，将取节点逻辑改为（单条 SQL 同时约束项目创建月 + 节点计划月）：

```python
def get_production_ranking(month: str) -> list[dict]:
    y, m = map(int, month.split("-"))
    ns = f"{y}-{m:02d}-01"
    ne = f"{y}-{m+1:02d}-01" if m < 12 else f"{y+1}-01-01"

    conn = get_connection()
    try:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT pnp.id, pnp.project_id, pnp.plan_qty, pnp.plan_date,
                       p.delivery_person
                FROM process_node_plans pnp
                JOIN projects p ON pnp.project_id = p.id
                WHERE pnp.process_name = '附件安装'
                  AND DATE_FORMAT(p.created_at, '%Y-%m') = %s
                  AND pnp.plan_date >= %s AND pnp.plan_date < %s
            """, (month, ns, ne))
            plans = [dict(r) for r in cur.fetchall()]
    finally:
        conn.close()

    node_ids = [p["id"] for p in plans]
    actuals = get_actuals_by_node_ids(node_ids)   # 既有批量函数

    agg = {}
    for p in plans:
        person = p["delivery_person"]
        if not person:
            continue
        a = agg.setdefault(person, {"plan": 0, "actual": 0, "projects": set()})
        a["plan"] += int(p["plan_qty"] or 0)
        a["actual"] += actuals.get(p["id"], 0)
        a["projects"].add(p["project_id"])

    rows = []
    for person, v in agg.items():
        rate = round(v["actual"] / v["plan"] * 100, 1) if v["plan"] else None
        rows.append({
            "delivery_person": person,
            "total_plan": v["plan"],
            "total_actual": v["actual"],
            "completion_rate": rate,
            "project_count": len(v["projects"]),
        })

    rows.sort(key=lambda r: (
        -(r["completion_rate"] if r["completion_rate"] is not None else -1),
        -r["total_plan"],
        r["delivery_person"],
    ))
    for i, r in enumerate(rows, 1):
        r["rank"] = i
    return rows
```

> `get_production_ranking(month)` 接口与 `get_production_ranking_detail(month, person)` 保持既有；仅前端改为传共享月份（见第六节）。

ensureMonth() {
  if (!this.filters.month) {

    this.filters.month = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`

  if (filters.keyword !== undefined) this.filters.keyword = filters.keyword
  if (filters.person !== undefined) this.filters.person = filters.person
  if (filters.status !== undefined) this.filters.status = filters.status
  if (filters.month !== undefined) this.filters.month = filters.month

  }

### 5. 导出逻辑（导出当前共享月份）

```js
async function onExport() {
  exporting.value = true
  try {
    const res = await fetchExportProjects(store.filters.month)
    const rows = (res.items || res.data?.items || [])
    const header = [['项目名称', '机型', '钢塔厂家', '截至上月进度', '本月计划', '整体进度(%)', '交付负责人']]
    const body = rows.map(r => [
      r.project_name, r.machine_type, r.factory_name,
      r.last_month_output, r.monthly_plan,
      (Number(r.progress_pct) || 0).toFixed(1), r.delivery_person,
    ])
    const ws = XLSX.utils.aoa_to_sheet([...header, ...body])
    ws['!cols'] = [
      { wch: 26 }, { wch: 14 }, { wch: 18 },
      { wch: 14 }, { wch: 12 }, { wch: 14 }, { wch: 14 },
    ]
    const wb = XLSX.utils.book_new()
    XLSX.utils.book_append_sheet(wb, ws, '生产进度总览')
    const buf = XLSX.write(wb, { bookType: 'xlsx', type: 'array' })
    saveAs(new Blob([buf], { type: 'application/octet-stream' }), `生产进度总览_${store.filters.month}.xlsx`)
    ElMessage.success('导出成功')
  } catch (e) {
    ElMessage.error('导出失败，请重试')
  } finally {
    exporting.value = false
  }
}
```

## 七、数据口径对照表（导出 7 字段）

| 导出字段 | 来源 | 说明 |
|---|---|---|
| 项目名称 | `project_name` | 直接取 |
| 机型 | `machine_type` | 直接取 |
| 钢塔厂家 | `factory_name` | 直接取 |
| 截至上月进度 | `last_month_output` | 调度令导入时写入的"截止上月月底出品" |
| 本月计划 | `monthly_plan` | 调度令导入时写入的本月计划出品量 |
| 整体进度 | `progress_pct` | 后端实时算（附件安装 实际/计划 ×100，1 位小数） |
| 交付负责人 | `delivery_person` | 直接取 |

导出仅包含 `store.filters.month` 所对应月份（`created_at` 年月匹配）的项目，与三页列表/KPI 口径完全一致。

## 十、可选增强（二期）

2. **月份范围选择**：升级为 `type="monthrange"` 支持多月份合并导出。
